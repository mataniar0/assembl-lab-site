import json, sys
from pathlib import Path
root=Path("/workspace/assembl-lab-site")
sys.path.insert(0,str(root/"tests"))
import pre_push_integration as proof
original=proof.Proof.report_for
def retain_report(self,checkout,commit):
    report=original(self,checkout,commit)
    out=root/".qa-reports/al-001/fixture-reports"
    out.mkdir(parents=True,exist_ok=True)
    (out/(commit+".json")).write_text(json.dumps(report,ensure_ascii=False,indent=2)+"\n")
    return report
proof.Proof.report_for=retain_report
sys.argv=[str(root/"tests/pre_push_integration.py"),"--browser","/usr/bin/chromium","--report",str(root/".qa-reports/al-001/pre-push-integration.json")]
sys.exit(proof.main())
